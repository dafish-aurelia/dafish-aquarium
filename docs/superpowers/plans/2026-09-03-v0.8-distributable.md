```javascript
(function (global) {
  'use strict';
  // URL 脱敏：丢弃 query 与 fragment，仅保留 origin + path。
  // 信封里只允许出现这里的产物（隐私边界，见设计文档 §4.1）。
  function sanitizeUrl(rawUrl) {
    try {
      const u = new URL(rawUrl);
      return { url: u.origin + u.pathname, origin: u.origin, domain: u.hostname };
    } catch (e) {
      return { url: '', origin: '', domain: '' };
    }
  }
  // 水缸主页（v0.8 起不再硬编码出厂路径）：
  // 默认空串 = 珊瑚礁页（newtab.html）就是家；
  // 用户在设置页显式填 storage.home_url 才跳外部页面（file:// 或 https://）。
  const DEFAULT_HOME = '';
  global.DafeiyuSanitize = {
    sanitizeUrl,
    HOME_URL: DEFAULT_HOME,
    DEFAULT_HOME,
    setHomeUrl(u) { if (u && typeof u === 'string') this.HOME_URL = u; },
  };
  // 各上下文启动时拉一次覆盖值（storage 不可达时静默用默认）
  try {
    chrome.storage.local.get('home_url').then(({ home_url }) => {
      if (home_url) global.DafeiyuSanitize.setHomeUrl(home_url);
    }).catch(() => {});
    chrome.storage.onChanged.addListener((ch, area) => {
      if (area === 'local' && ch.home_url) global.DafeiyuSanitize.setHomeUrl(ch.home_url.newValue);
    });
  } catch (e) { /* 非扩展上下文（如 node 测试） */ }
  if (typeof module !== 'undefined' && module.exports) {
    module.exports = { sanitizeUrl };
  }
})(typeof window !== 'undefined' ? window : globalThis);

**设计决策（先定后写）**：
1. 安装器以**扩展仓库根目录**为锚点（`install.py` 所在目录的父目录），用户把仓库 clone 到哪都行——不做全局安装、不拷贝到 Program Files，符合"绿色软件"习惯；
2. native-host manifest（`dafeiyu_gatekeeper.json`）与 bat 由安装器**按实际路径生成**到 `native-host/generated/`（gitignore），注册表指向生成文件；仓库里的 `dafeiyu_gatekeeper.json` 保留为**只读模板**（供 Level 2 用户参考），不再直接注册；
3. **bat 内容用 8.3 短路径 + 纯 ASCII**：`cmd /c for %I in (path) do @echo %~sI` 取短路径，规避中文用户名/中文路径在非 UTF-8 代码页机器上的解析失败（这正是本机当前 bat 的坑）；短路径取不到时回退长路径并在安装日志里警告；
4. `install.bat` 探测 python 优先级：`py -3` → `python` → `python3`，全都没有则提示安装 Python 3.10+ 并中止；**不自动下载**（避免依赖网络与脚本安全争议）；
5. Level 0/1 用户**不需要 native messaging**（alarm 看门狗已能保活信局），安装器对 native-host 注册失败只警告不中止——扩展本身照常工作；
6. Python 侧不依赖 venv：直接调用探测到的系统 python 跑 `server/pet_mailbox.py`（stdlib-only）。原工作区 `.venv` 用户仍可手动指定 `--python <path>` 覆盖。

# -*- coding: utf-8 -*-
"""大肥鱼扩展安装器：按用户实际路径生成 native-host 配置并注册。
绿色安装：一切产物落在仓库 native-host/generated/ 与 HKCU 注册表，
不写系统目录、不需要管理员。卸载 = uninstall.py 一键还原。
"""
import argparse
import base64
import json
import os
import shutil
import subprocess
import sys
from pathlib import Path

PINNED_EXT_ID = 'bekeaffohpngiaidlffmfmjohjflmail'  # manifest key 钉死（Task 3）
NM_HOST_NAME = 'dafeiyu_gatekeeper'
REG_KEY = rf'Software\Google\Chrome\NativeMessagingHosts\{NM_HOST_NAME}'
EXT_ROOT = Path(__file__).resolve().parents[1]


def get_short_path(path):
    """8.3 短路径：cmd 的 %~sI 换算。取不到回退原路径（调用方须保证 ASCII 校验）。"""
    try:
        r = subprocess.run(
            ['cmd', '/c', 'for %I in ("{}") do @echo %~sI'.format(path)],
            capture_output=True, text=True, timeout=10)
        sp = r.stdout.strip().splitlines()[-1].strip()
        return sp if sp and Path(sp).exists() else str(path)
    except Exception:
        return str(path)


def find_python():
    """探测可用 python：py -3 → python → python3。全无则返回 None。"""
    for cmd in (['py', '-3'], ['python'], ['python3']):
        try:
            r = subprocess.run(cmd + ['-c', 'import sys; print(sys.executable)'],
                               capture_output=True, text=True, timeout=10)
            exe = r.stdout.strip().splitlines()[-1]
            if r.returncode == 0 and exe:
                return exe
        except Exception:
            continue
    return None


def render_bat(python_exe, host_script):
    """生成纯 ASCII bat。python/脚本路径先转 8.3 短路径防中文代码页问题。"""
    py_s = get_short_path(python_exe)
    host_s = get_short_path(host_script)
    for p in (py_s, host_s):
        if not p.isascii():
            raise RuntimeError(f'短路径不可用且原路径含非 ASCII 字符: {p!r} —— '
                               '请在含中文的路径上启用 8.3 命名或把仓库放在纯英文路径')
    return '@echo off\r\n"{}" "{}"\r\n'.format(py_s, host_s)


def render_manifest(bat_path):
    return json.dumps({
        'name': NM_HOST_NAME,
        'description': '鲸鱼娘后勤看门人：寄生 Chrome，确保信局随浏览器起落',
        'path': bat_path,
        'type': 'stdio',
        'allowed_origins': [f'chrome-extension://{PINNED_EXT_ID}/'],
    }, ensure_ascii=False, indent=2)


def short_path_compat_ok():
    """FAT32/注册表关闭 8.3 时告警（不中止）。"""
    r = subprocess.run(['fsutil', 'behavior', 'query', 'disable8dot3'],
                       capture_output=True, text=True)
    return '1' not in (r.stdout or '').split(':')[-1].strip()


def main():
    ap = argparse.ArgumentParser(description='大肥鱼扩展安装器')
    ap.add_argument('--python', help='指定 python.exe（默认自动探测）')
    ap.add_argument('--registry', action='store_true',
                    help='真的写注册表（默认只生成文件，打印将要写的键值）')
    args = ap.parse_args()

    python_exe = args.python or find_python()
    if not python_exe:
        print('[install] 未找到 Python 3.10+，请安装后重试：https://www.python.org/')
        return 1
    print(f'[install] python: {python_exe}')

    host_script = EXT_ROOT / 'native-host' / 'dafeiyu_gatekeeper_host.py'
    if not host_script.exists():
        # 分发版：仓库根可能没有 workspace/scripts —— 用扩展自己的轻量看护
        host_script = EXT_ROOT / 'native-host' / 'gatekeeper_host_lite.py'

    gen_dir = EXT_ROOT / 'native-host' / 'generated'
    gen_dir.mkdir(parents=True, exist_ok=True)
    bat_path = gen_dir / 'dafeiyu_gatekeeper.bat'
    bat_path.write_text(render_bat(python_exe, str(host_script)), encoding='ascii', newline='')
    mf_path = gen_dir / 'dafeiyu_gatekeeper.json'
    mf_path.write_text(render_manifest(str(bat_path)), encoding='utf-8')
    print(f'[install] native-host 产物: {gen_dir}')

    if args.registry:
        abs_mf = str(mf_path.resolve())

                           '/ve', '/t', 'REG_SZ', '/d', abs_mf, '/f'],
                          capture_output=True, text=True)
        if r.returncode == 0:
            print(f'[install] ✓ 注册表已写入 HKCU\\{REG_KEY}')
        else:
            print(f'[install] ⚠ 注册表写入失败: {r.stderr.strip()}')
            return 1
    else:
        print(f'[install] 预演模式。写入注册表请加 --registry:')
        print(f'  reg add HKCU\\{REG_KEY} /ve /t REG_SZ /d "{mf_path.resolve()}" /f')
    return 0

    sys.exit(main())

# -*- coding: utf-8 -*-
"""轻量看护宿主（分发版）：只实现 Native Messaging 协议应答，
信局保活改由扩展侧 pet-gate alarm 驱动（alarm 唤醒 SW → connectNative 拉起本进程 →
应答 pong）。不依赖 workspace 的 start_dafeiyu.py。
协议: 4 字节 LE 长度前缀 + UTF-8 JSON。stdout 只准写帧，日志全走 stderr。
"""
import json
import os
import struct
import subprocess
import sys
import time

EXT_ROOT = Path = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
sys.path.insert(0, os.path.join(EXT_ROOT, 'server'))

MAILBOX = os.path.join(EXT_ROOT, 'server', 'pet_mailbox.py')
DATA_DIR = os.path.join(os.environ.get('LOCALAPPDATA', os.path.expanduser('~')),
                        'dafeiyu', 'pet-mailbox')

_PROTO = sys.stdout
sys.stdout = sys.stderr


def _read_message():
    raw = sys.stdin.buffer.read(4)
    if not raw or len(raw) < 4:
        return None
    (length,) = struct.unpack('<I', raw)
    if length <= 0 or length > 1_000_000:
        return None
    body = sys.stdin.buffer.read(length)
    if not body or len(body) < length:
        return None
    return body


def _send_message(obj):
    data = json.dumps(obj, ensure_ascii=False).encode('utf-8')
    _PROTO.buffer.write(struct.pack('<I', len(data)))
    _PROTO.buffer.write(data)
    _PROTO.buffer.flush()


def _mailbox_alive():
    import urllib.request
    try:
        urllib.request.urlopen('http://127.0.0.1:13140/api/token', timeout=2)
        return True
    except Exception:
        return False


def _ensure_mailbox(python_exe):
    if _mailbox_alive():
        return
    subprocess.Popen([python_exe, MAILBOX],
                     creationflags=0x08000000,  # NO_WINDOW
                     stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)


def main():
    python_exe = sys.executable
    first = True
    while True:
        msg = _read_message()
        if msg is None:
            return
        try:
            _ensure_mailbox(python_exe)
        except Exception:
            pass
        _send_message({'type': 'pong', 'ok': True, 'pid': os.getpid(),
                      'ts': int(time.time() * 1000), 'first': first})
        first = False


if __name__ == '__main__':
    main()
```
